One SDK, every tool

import { Composio } from "composio";   const client = new Composio({ apiKey: process.env.COMPOSIO_API_KEY, });   // Execute a tool with one call const result = await client.execute({ tool: "github.create_issue", params: { repo: "acme/api", title: "Fix auth bug" }, entityId: "agent-prod-01", });   console.log(result.data.html_url); // → https://github.com/acme/api/issues/293
