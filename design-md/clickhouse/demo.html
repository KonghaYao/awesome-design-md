SELECT
    toStartOfHour(timestamp) AS hour,
    count() AS events,
    avg(response_time) AS avg_ms,
    quantile(0.95)(response_time) AS p95
FROM analytics.events
WHERE timestamp BETWEEN
    '2025-01-01' AND '2025-01-31'
  AND event_type = 'page_view'
GROUP BY hour
ORDER BY hour DESC
LIMIT 100

-- Vector search: find similar support tickets
WITH target_embedding AS (
    SELECT embed('database timeout error') AS vec
)
SELECT
    ticket_id,
    summary,
    cosineDistance(embedding, target.vec) AS similarity
FROM support_tickets, target_embedding AS target
ORDER BY similarity ASC
LIMIT 5

-- Real-time funnel analysis
SELECT
    funnel_step,
    uniqExact(user_id) AS users,
    formatReadableQuantity(users) AS display
FROM funnel_events
WHERE event_date >= today() - 7
GROUP BY funnel_step
ORDER BY funnel_step
